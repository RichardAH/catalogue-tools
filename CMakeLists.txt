cmake_minimum_required(VERSION 3.16)
project(catl CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
option(CATL_TLS "wss:// and https:// support via OpenSSL" ON)

find_package(Threads REQUIRED)
add_library(catlcore STATIC
  src/sha.cpp src/zlib.cpp src/catl.cpp src/index.cpp src/shamap.cpp
  src/protocol.cpp src/net.cpp src/ops.cpp src/fetch.cpp src/tui.cpp)
target_include_directories(catlcore PUBLIC src)
target_link_libraries(catlcore PUBLIC Threads::Threads)
target_compile_options(catlcore PRIVATE -Wall -Wextra)
if(CATL_TLS)
  find_package(OpenSSL)
  if(OPENSSL_FOUND)
    target_compile_definitions(catlcore PUBLIC CATL_WITH_OPENSSL)
    target_link_libraries(catlcore PUBLIC OpenSSL::SSL OpenSSL::Crypto)
  else()
    message(WARNING "OpenSSL not found: building without TLS (wss/https unavailable)")
  endif()
endif()

add_executable(catl src/main.cpp)
target_link_libraries(catl PRIVATE catlcore)
add_executable(catl_tests tests/test_main.cpp)
target_link_libraries(catl_tests PRIVATE catlcore)
enable_testing()
add_test(NAME catl_tests COMMAND catl_tests)
install(TARGETS catl RUNTIME DESTINATION bin)
